# GET: картинки немає в кеші (X-Image-Source: http.cat)
curl -s -D - -o /tmp/200.jpg localhost:3000/200

# GET: повторний запит (X-Image-Source: cache)
curl -s -D - -o /tmp/200.jpg localhost:3000/200

# GET: неіснуючий код (404)
curl -s -D - localhost:3000/999

# GET: некоректний шлях (404)
curl -s -D - localhost:3000/abc

# Недозволений метод (405 + Allow)
curl -s -D - -X POST localhost:3000/200

# PUT: записати картинку в кеш (201)
curl -s -D - -X PUT --data-binary @/tmp/200.jpg localhost:3000/418

# GET після PUT: віддається наша картинка (X-Image-Source: cache)
curl -s -D - -o /tmp/418.jpg localhost:3000/418

# PUT з порожнім тілом (400)
curl -s -D - -X PUT localhost:3000/418

# DELETE: видалити картинку з кешу (200)
curl -s -D - -X DELETE localhost:3000/418

# DELETE: картинки вже немає (404)
curl -s -D - -X DELETE localhost:3000/418
